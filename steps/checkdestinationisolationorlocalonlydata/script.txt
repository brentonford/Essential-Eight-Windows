$destinationPath = "{backupDestinationPath}".Trim()
$thresholdMb     = 100

$os       = Get-CimInstance -ClassName Win32_OperatingSystem
$isServer = $os.Caption -match "Windows Server"

if (-not $isServer) {
    Write-Output "CHECK Destination Isolation : SKIP -- not applicable on desktop OS"
    exit 0
}

if ([string]::IsNullOrWhiteSpace($destinationPath)) {
    Write-Output "CHECK Destination Isolation : WARN -- no backup destination path parameter provided"
    exit 1
}

Write-Output "Backup destination path : $destinationPath"

$destExists = Test-Path $destinationPath

Write-Output "Destination exists      : $destExists"

if (-not $destExists) {
    Write-Output "CHECK Destination Isolation : FAIL -- destination path does not exist"
    exit 1
}

# Check destination is not on the system drive (isolation check)
$systemDrive = $env:SystemDrive.TrimEnd('\')
$destDrive   = (Split-Path $destinationPath -Qualifier).TrimEnd('\')
$isIsolated  = $destDrive -ne $systemDrive

Write-Output "System drive            : $systemDrive"
Write-Output "Destination drive       : $destDrive"
Write-Output "Drive isolated          : $isIsolated"

# Check destination size
$destItems = Get-ChildItem -Path $destinationPath -Recurse -ErrorAction SilentlyContinue |
    Where-Object { -not $_.PSIsContainer }

$sizeBytes = 0
foreach ($item in $destItems) {
    $sizeBytes += $item.Length
}

$sizeMb = [math]::Round($sizeBytes / 1MB, 1)

Write-Output "Destination size (MB)   : $sizeMb"
Write-Output "Threshold (MB)          : $thresholdMb"

$sizeOk = $sizeMb -gt $thresholdMb

Write-Output "CHECK Destination Size      : $(if ($sizeOk) { 'PASS' } else { 'FAIL -- destination below threshold (no backups present?)' })"
Write-Output "CHECK Destination Isolation : $(if ($isIsolated) { 'PASS' } else { 'WARN -- backup on system drive' })"
exit $(if ($sizeOk -and $isIsolated) { 0 } else { 1 })
