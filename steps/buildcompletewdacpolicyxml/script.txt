# Writes the complete WDAC policy XML as a literal here-string.
# Avoids all XML DOM manipulation, which causes namespace decoration
# issues that the ConvertFrom-CIPolicy and Set-RuleOption validators reject.
#
# Structure: AllowAll base (ID_ALLOW_ALL_1/2) + FilePathRule deny rules (ISM-1870).
# Deny rules take precedence over the allow-all entries.
# FileRulesRef sits directly inside ProductSigners per the SiPolicy schema.

$archivePath     = "{policyExportPath}"
$policyName      = "E8-ML1-AppControl"
$policyXmlPath   = Join-Path $archivePath "$policyName.xml"
$enforcementMode = "{wdacEnforcementMode}".Trim().ToLower()
$allowAllPath    = "$env:windir\schemas\CodeIntegrity\ExamplePolicies\AllowAll.xml"

if (-not (Test-Path $allowAllPath)) {
    Write-Error "AllowAll template not found: $allowAllPath"
    exit 1
}

# Generate path rules via the cmdlet — the only way to produce schema-valid IDs.
# New-CIPolicyRule -FilePathRule does not support -Deny, so rules are created as
# Allow and converted to Deny after the merge.
$pathRules  = @()
$pathRules += New-CIPolicyRule -FilePathRule "%OSDrive%\Users\*\AppData\Local\Temp\*"
$pathRules += New-CIPolicyRule -FilePathRule "%OSDrive%\Users\*\AppData\Local\*"
$pathRules += New-CIPolicyRule -FilePathRule "%OSDrive%\Users\*\AppData\Roaming\*"
$pathRules += New-CIPolicyRule -FilePathRule "%OSDrive%\Users\*\Downloads\*"

Write-Output "Path rules generated: $($pathRules.Count)"
$pathRules | ForEach-Object { Write-Output "  $($_.Id)  $($_.FilePathRule)" }

# Merge path rules into AllowAll — Merge-CIPolicy handles FileRulesRef placement correctly
Merge-CIPolicy -PolicyPaths $allowAllPath -OutputFilePath $policyXmlPath -Rules $pathRules
Write-Output "Policy merged: $policyXmlPath"

# Flip Action from Allow to Deny on FilePathRule elements only.
# Allow elements use FileName not Action, so this replacement is safe.
$content = Get-Content -Path $policyXmlPath -Raw
$content = $content -replace '(<FilePathRule\s[^>]+)Action="Allow"', '$1Action="Deny"'
[System.IO.File]::WriteAllText($policyXmlPath, $content, [System.Text.UTF8Encoding]::new($false))
Write-Output "FilePathRule actions set to Deny."

# Set UMCI and enforcement mode — Set-RuleOption now operates on a valid policy
Set-RuleOption -FilePath $policyXmlPath -Option 0

if ($enforcementMode -eq "enforce") {
    Set-RuleOption -FilePath $policyXmlPath -Option 3 -Delete
    Write-Output "Enforcement: Enforce"
    Write-Output "WARNING: Unauthorised execution in blocked paths will be prevented."
    Write-Output "Run Audit mode first and review Event ID 3076 before enforcing in production."
} else {
    Set-RuleOption -FilePath $policyXmlPath -Option 3
    Write-Output "Enforcement: Audit -- review Event ID 3076 in Microsoft-Windows-CodeIntegrity/Operational"
}

# Assign policy name and a new stable GUID
Set-CIPolicyIdInfo -FilePath $policyXmlPath -PolicyName $policyName -ResetPolicyID

$policyId = ([xml](Get-Content -Path $policyXmlPath -Raw)).SiPolicy.PolicyID
Write-Output ""
Write-Output "Policy ID  : $policyId"
Write-Output "Policy XML : $policyXmlPath"
Write-Output "Mode       : $enforcementMode"
