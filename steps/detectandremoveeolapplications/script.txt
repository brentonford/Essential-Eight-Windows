# ISM-1704 (ML1): office productivity suites, web browsers, email clients, PDF software,
# Adobe Flash Player, and security products no longer supported by vendors must be removed.

$eolApplications = @(
    @{ Name = "Adobe Flash Player";          WingetId = $null },
    @{ Name = "Internet Explorer";           WingetId = $null },
    @{ Name = "Java 6";                      WingetId = $null },
    @{ Name = "Java 7";                      WingetId = $null },
    @{ Name = "Java(TM) 6";                  WingetId = $null },
    @{ Name = "Java(TM) 7";                  WingetId = $null },
    @{ Name = "Microsoft Silverlight";       WingetId = $null },
    @{ Name = "Microsoft Visual C++ 2005";   WingetId = $null },
    @{ Name = "Microsoft Visual C++ 2008";   WingetId = $null },
    @{ Name = "Python 2";                    WingetId = $null },
    @{ Name = "7-Zip 1";                     WingetId = $null },
    @{ Name = "7-Zip 2";                     WingetId = $null }
)

$registryPaths = @(
    "HKLM:\Software\Microsoft\Windows\CurrentVersion\Uninstall\*",
    "HKLM:\Software\Wow6432Node\Microsoft\Windows\CurrentVersion\Uninstall\*"
)

$installedApps = $registryPaths | ForEach-Object {
    Get-ItemProperty -Path $_ -ErrorAction SilentlyContinue
} | Where-Object { $_.DisplayName }

Write-Output "Installed applications scanned: $($installedApps.Count)"

$foundEol = @()
foreach ($eolEntry in $eolApplications) {
    $matchedApps = $installedApps | Where-Object { $_.DisplayName -like "*$($eolEntry.Name)*" }
    foreach ($match in $matchedApps) {
        Write-Output "EOL APPLICATION FOUND: $($match.DisplayName) (version: $($match.DisplayVersion))"
        $foundEol += $match
    }
}

if ($foundEol.Count -eq 0) {
    Write-Output "No EOL applications detected."
    exit 0
}

Write-Output "Attempting removal of $($foundEol.Count) EOL application(s)..."

$removalFailed = $false

foreach ($app in $foundEol) {
    Write-Output "Removing: $($app.DisplayName)..."

    if (-not $app.UninstallString) {
        Write-Error "  No uninstall string found for: $($app.DisplayName). Manual removal required."
        $removalFailed = $true
        continue
    }

    $uninstallCmd = $app.UninstallString -replace '/I', '/X'

    if ($uninstallCmd -match "msiexec") {
        $productCode = ($uninstallCmd | Select-String -Pattern "\{[A-Z0-9\-]+\}").Matches.Value
        if ($productCode) {
            Start-Process -FilePath "msiexec.exe" `
                -ArgumentList "/X $productCode /qn /norestart" `
                -Wait `
                -NoNewWindow
            Write-Output "  Removal attempted via msiexec: $($app.DisplayName)"
            continue
        }
    }

    Write-Error "  Automatic removal not possible for: $($app.DisplayName)"
    Write-Error "  Uninstall string: $($app.UninstallString)"
    $removalFailed = $true
}

if ($removalFailed) {
    Write-Error "One or more EOL applications could not be removed automatically. Manual intervention required."
    exit 1
}
