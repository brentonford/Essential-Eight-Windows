$hostname        = $env:COMPUTERNAME
$reportDate      = Get-Date -Format "yyyy-MM-dd HH:mm:ss"
$destinationPath = "{backupDestinationPath}".Trim()
$retentionDays   = "{backupRetentionDays}".Trim()

# Feature check
$feature   = Get-WindowsFeature -Name "Windows-Server-Backup" -ErrorAction SilentlyContinue
$featureOk = $null -ne $feature -and $feature.InstallState -eq "Installed"

# Schedule check via Task Scheduler
$task = Get-ScheduledTask -TaskPath "\Microsoft\Windows\Backup\" -ErrorAction SilentlyContinue |
        Where-Object { $_.TaskName -match "WindowsBackup" } |
        Select-Object -First 1
$scheduleOk = $null -ne $task

# Backup version check via wbadmin get versions
$versionsOutput = cmd /c "wbadmin get versions" 2>&1
$versionLines   = $versionsOutput | Where-Object { $_ -match "Version identifier:" }
$backupSetCount = @($versionLines).Count

# Parse most recent backup time
$lastBackupLine = $versionsOutput | Where-Object { $_ -match "Backup time:" } | Select-Object -Last 1
$lastSuccess    = $null
$ageHours       = $null
if ($lastBackupLine) {
    try {
        $dateStr     = ($lastBackupLine -replace "Backup time:\s*", "").Trim()
        $lastSuccess = [datetime]::Parse($dateStr)
        $ageHours    = [math]::Round((New-TimeSpan -Start $lastSuccess -End (Get-Date)).TotalHours, 1)
    } catch { }
}
$backupAgeOk = $null -ne $ageHours -and $ageHours -le 24

# Destination isolation check
$destIsUNC     = $destinationPath -match '^(\\\\|//)'
$ism1811Result = if ($destIsUNC) { "PASS" } elseif (-not [string]::IsNullOrWhiteSpace($destinationPath)) { "GAP -- local destination" } else { "NOT CONFIGURED" }

$overallPass = $featureOk -and $scheduleOk -and $backupAgeOk -and $backupSetCount -gt 0

$separator = "=" * 60

Write-Output $separator
Write-Output "  ESSENTIAL EIGHT ML1 -- REGULAR BACKUPS"
Write-Output "  ASD Control: Mitigation Strategy 8"
Write-Output "  ISM Controls: 1511, 1515, 1810, 1811, 1812, 1814"
Write-Output "  Reference  : cyber.gov.au/essential-eight"
Write-Output $separator
Write-Output "  Server     : $hostname"
Write-Output "  Report Date: $reportDate"
Write-Output "  Generated  : AttuneOps"
Write-Output $separator
Write-Output ""
Write-Output "  BACKUP INFRASTRUCTURE"
Write-Output "    WSB Feature installed : $(if ($featureOk) { 'Yes' } else { 'No' })"
Write-Output "    Backup schedule set   : $(if ($scheduleOk) { 'Yes' } else { 'No' })"
Write-Output "    Backup sets found     : $backupSetCount"
Write-Output ""
Write-Output "  [ISM-1511] Backup Currency"
Write-Output "    Last successful backup : $(if ($null -ne $lastSuccess) { $lastSuccess.ToString('yyyy-MM-dd HH:mm:ss') } else { 'Never / Unknown' })"
Write-Output "    Age (hours)            : $(if ($null -ne $ageHours) { $ageHours } else { 'N/A' })"
Write-Output "    Within 24-hour SLA     : $(if ($backupAgeOk) { 'Yes' } else { 'No' })"
Write-Output "    Configured retention   : $(if ([string]::IsNullOrWhiteSpace($retentionDays)) { 'Not set' } else { $retentionDays }) days"
Write-Output ""
Write-Output "  [ISM-1811] Destination Isolation"
Write-Output "    Destination path : $(if ([string]::IsNullOrWhiteSpace($destinationPath)) { 'Not configured' } else { $destinationPath })"
Write-Output "    Type             : $(if ($destIsUNC) { 'UNC network share' } else { 'Local volume' })"
Write-Output "    Result           : $ism1811Result"
if ($ism1811Result -ne "PASS") {
    Write-Output "    ACTION: Configure a UNC path on a separate backup server for ISM-1811 compliance."
}
Write-Output ""
Write-Output "  [ISM-1515] Restoration Tested"
Write-Output "    Backup versions  : $backupSetCount available"
Write-Output "    Full DR exercise : REQUIRES PROCESS DOCUMENTATION"
Write-Output "    ACTION: Conduct and document a full DR exercise at least annually."
Write-Output ""
Write-Output $separator
Write-Output "  OVERALL STATUS : $(if ($overallPass) { 'COMPLIANT' } else { 'NON-COMPLIANT' })"
Write-Output $separator

if (-not $overallPass) {
    Write-Output ""
    Write-Output "  REMEDIATION: Run the Configure Regular Backups - ML1 Blueprint"
    Write-Output "  against this server and address any gaps above."
    exit 1
}
