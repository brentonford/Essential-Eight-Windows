# WinRM is the remote management transport used by AttuneOps and PowerShell remoting.
# Basic Authentication transmits credentials as base64-encoded plain text — it must
# be disabled. Kerberos (domain-joined) or NTLM with NTLMv2 (workgroup) are acceptable.
#
# This step hardens the WinRM listener to disallow Basic and Digest auth,
# require HTTPS or Kerberos/NTLM, and enforce NTLMv2-minimum authentication.

$winrmAuthPaths = @{
    Basic     = "WSMan:\localhost\Service\Auth\Basic"
    Digest    = "WSMan:\localhost\Service\Auth\Digest"
    Kerberos  = "WSMan:\localhost\Service\Auth\Kerberos"
    Negotiate = "WSMan:\localhost\Service\Auth\Negotiate"
}

$disableAuth = @("Basic", "Digest")
$enableAuth  = @("Kerberos", "Negotiate")

foreach ($name in $disableAuth) {
    $path = $winrmAuthPaths[$name]
    $existing = Get-Item -Path $path -ErrorAction SilentlyContinue
    if ($null -eq $existing) {
        Write-Output "WinRM $name : path not present — treating as disabled, skipping."
        continue
    }
    Set-Item -Path $path -Value $false
    Write-Output "WinRM $name : set to false (disabled)"
}

foreach ($name in $enableAuth) {
    $path = $winrmAuthPaths[$name]
    $existing = Get-Item -Path $path -ErrorAction SilentlyContinue
    if ($null -eq $existing) {
        Write-Output "WinRM $name : path not present — skipping."
        continue
    }
    Set-Item -Path $path -Value $true
    Write-Output "WinRM $name : set to true (enabled)"
}

$lmPath = "HKLM:\SYSTEM\CurrentControlSet\Control\Lsa"
Set-ItemProperty -Path $lmPath -Name "LmCompatibilityLevel" -Value 5 -Type DWord
Write-Output ""
Write-Output "LAN Manager authentication level set to 5 (NTLMv2 only, refuse LM and NTLM)."

Set-ItemProperty -Path $lmPath -Name "NoLMHash" -Value 1 -Type DWord
Write-Output "LM hash storage disabled (NoLMHash = 1)."

Write-Output ""
Write-Output "WinRM authentication hardened."
Write-Output "NOTE: If this server uses workgroup authentication (non-domain), verify"
Write-Output "that Negotiate (NTLM) is still functional before relying on WinRM connections."
