# WinRM is the remote management transport used by AttuneOps and PowerShell remoting.
# Basic Authentication transmits credentials as base64-encoded plain text -- it must
# be disabled. Kerberos (domain-joined) or NTLM with NTLMv2 (workgroup) are acceptable.
#
# This step hardens the WinRM listener to disallow Basic and Digest auth,
# require HTTPS or Kerberos/NTLM, and enforce NTLMv2-minimum authentication.

# Disable Basic Authentication for WinRM service (server side)
Set-Item -Path "WSMan:\localhost\Service\Auth\Basic"     -Value $false
$basicVal = (Get-Item -Path "WSMan:\localhost\Service\Auth\Basic" -ErrorAction SilentlyContinue).Value
Write-Output "WinRM Basic    : set to $basicVal (disabled)"

# Digest path is absent on Server 2022 when not explicitly enabled -- treat as disabled
$digestPath = "WSMan:\localhost\Service\Auth\Digest"
if (Test-Path $digestPath) {
    Set-Item -Path $digestPath -Value $false
    $digestVal = (Get-Item -Path $digestPath -ErrorAction SilentlyContinue).Value
    Write-Output "WinRM Digest   : set to $digestVal (disabled)"
} else {
    Write-Output "WinRM Digest   : path not present -- treating as disabled, skipping."
}

Set-Item -Path "WSMan:\localhost\Service\Auth\Kerberos"  -Value $true
$kerberosVal = (Get-Item -Path "WSMan:\localhost\Service\Auth\Kerberos" -ErrorAction SilentlyContinue).Value
Write-Output "WinRM Kerberos : set to $kerberosVal (enabled)"

Set-Item -Path "WSMan:\localhost\Service\Auth\Negotiate" -Value $true
$negotiateVal = (Get-Item -Path "WSMan:\localhost\Service\Auth\Negotiate" -ErrorAction SilentlyContinue).Value
Write-Output "WinRM Negotiate: set to $negotiateVal (enabled)"

# Enforce NTLMv2 minimum via LAN Manager Authentication Level
$lmPath = "HKLM:\SYSTEM\CurrentControlSet\Control\Lsa"
# LmCompatibilityLevel values:
#   0-1 = LM + NTLM allowed (insecure)
#   2   = NTLM only (no LM)
#   3   = NTLMv2 only
#   4   = NTLMv2 only, refuse LM from clients
#   5   = NTLMv2 only, refuse LM and NTLM from clients
# Value 5 is most secure but may break legacy clients. Value 3 is recommended minimum.
Set-ItemProperty -Path $lmPath -Name "LmCompatibilityLevel" -Value 5 -Type DWord
Write-Output ""
Write-Output "LAN Manager authentication level set to 5 (NTLMv2 only, refuse LM and NTLM)."

# Disable NTLM hash storage (NoLMHash)
Set-ItemProperty -Path $lmPath -Name "NoLMHash" -Value 1 -Type DWord
Write-Output "LM hash storage disabled (NoLMHash = 1)."

Write-Output ""
Write-Output "WinRM authentication hardened."
Write-Output "NOTE: If this server uses workgroup authentication (non-domain), verify"
Write-Output "that Negotiate (NTLM) is still functional before relying on WinRM connections."
