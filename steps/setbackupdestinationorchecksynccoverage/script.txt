$os       = Get-CimInstance -ClassName Win32_OperatingSystem
$isServer = $os.Caption -match "Windows Server"

if ($isServer) {
    Import-Module WindowsServerBackup

    $destinationPath = "{backupDestinationPath}".Trim()

    if ($destinationPath -notmatch '^\\\\') {
        Write-Error "Backup destination must be a UNC path. Provided: $destinationPath"
        Write-Error "A local volume does not meet ISM-1811 isolation requirements."
        exit 1
    }

    if (-not (Test-Path $destinationPath)) {
        Write-Error "Backup destination not reachable: $destinationPath"
        exit 1
    }

    Write-Output "Destination (UNC) reachable: $destinationPath"

    # Isolation check
    $testFile    = Join-Path $destinationPath "isolation_check_$(Get-Random).tmp"
    $canWrite    = $false
    try {
        [System.IO.File]::WriteAllText($testFile, "test")
        Remove-Item $testFile -Force -ErrorAction SilentlyContinue
        $canWrite = $true
        Write-Output "WARN ISM-1811: Current credential has write access to destination."
        Write-Output "Restrict share permissions to backup service account only."
    } catch {
        Write-Output "Isolation check PASSED: current credential cannot write to destination."
    }

    $policy        = Get-WBPolicy
    $networkTarget = New-WBBackupTarget -NetworkPath $destinationPath
    Add-WBBackupTarget -Policy $policy -Target $networkTarget -Force
    Set-WBPolicy -Policy $policy
    Write-Output "Backup destination set."
    exit 0
}

# Desktop: check which user profile locations are NOT covered by OneDrive sync
Write-Output "Checking for local-only data in user profile..."

$oneDriveRoot = (Get-ItemProperty -Path "HKCU:\SOFTWARE\Microsoft\OneDrive" -Name "UserFolder" -ErrorAction SilentlyContinue).UserFolder
$userProfile  = $env:USERPROFILE
$thresholdMb  = 100

$profileDirs = Get-ChildItem -Path $userProfile -Directory -ErrorAction SilentlyContinue |
    Where-Object { $_.Name -notin @("AppData", "Searches", "Links", "Contacts", "Saved Games", "SendTo") }

$localOnlyDirs = @()

foreach ($dir in $profileDirs) {
    $isUnderOneDrive = $null -ne $oneDriveRoot -and $dir.FullName.StartsWith($oneDriveRoot)

    if ($isUnderOneDrive) {
        Write-Output "SYNCED    : $($dir.FullName)"
        continue
    }

    $sizeBytes = (Get-ChildItem -Path $dir.FullName -Recurse -File -ErrorAction SilentlyContinue |
        Measure-Object -Property Length -Sum).Sum

    $sizeMb = [math]::Round(($sizeBytes ?? 0) / 1MB, 1)

    if ($sizeMb -gt $thresholdMb) {
        Write-Output "LOCAL ONLY: $($dir.FullName) -- ${sizeMb}MB (above ${thresholdMb}MB threshold)"
        $localOnlyDirs += "$($dir.Name) (${sizeMb}MB)"
    } else {
        Write-Output "local only: $($dir.FullName) -- ${sizeMb}MB (below threshold, acceptable)"
    }
}

if ($localOnlyDirs.Count -gt 0) {
    Write-Output ""
    Write-Output "WARN: $($localOnlyDirs.Count) large local-only director(y/ies) found."
    Write-Output "Consider moving these to OneDrive or a mapped network share to meet ISM-1511."
    $localOnlyDirs | ForEach-Object { Write-Output "  - $_" }
} else {
    Write-Output ""
    Write-Output "No large local-only directories found above ${thresholdMb}MB threshold."
}
