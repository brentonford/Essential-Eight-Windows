$hostname         = $env:COMPUTERNAME
$reportDate       = Get-Date -Format "yyyy-MM-dd HH:mm:ss"
$ml1ThresholdDays = 14

# [1] PSWindowsUpdate
$pswuModule = Get-Module -ListAvailable -Name PSWindowsUpdate
$pswuCheck  = if ($null -ne $pswuModule) { "PASS" } else { "FAIL" }

# [2] winget + scheduled task (workstation only)
$os            = Get-CimInstance -ClassName Win32_OperatingSystem
$isServerLocal = $os.Caption -match "Windows Server"
$wingetCmd     = Get-Command -Name winget -ErrorAction SilentlyContinue
$taskExists    = $null -ne (Get-ScheduledTask -TaskName "WinGet-AutoUpgrade" -ErrorAction SilentlyContinue)

$wingetCheck = if ($isServerLocal) {
    "SKIP (server)"
} elseif ($null -eq $wingetCmd) {
    "FAIL"
} elseif (-not $taskExists) {
    "FAIL -- WinGet-AutoUpgrade task missing"
} else {
    "PASS"
}

# [3] Outstanding updates
$outstandingUpdates = @()
$outstandingCheck   = "SKIP"
if ($null -ne $pswuModule) {
    Import-Module PSWindowsUpdate
    $outstandingUpdates = Get-WindowsUpdate `
        -Category "Application" `
        -NotTitle "Upgrade to" `
        -ErrorAction SilentlyContinue
    $outstandingCheck = if ($null -eq $outstandingUpdates -or $outstandingUpdates.Count -eq 0) {
        "PASS"
    } else {
        "FAIL"
    }
}

# [4] Update recency
# Do not filter on Categories -- it is a collection, not a string.
# -match against a collection does element-wise matching, not a boolean predicate.
# Take the most recent history entry directly.
$daysSinceUpdate = -1
$recencyCheck    = "SKIP"
$lastUpdateTitle = "Unknown"

if ($null -ne $pswuModule) {
    $mostRecent = Get-WUHistory -ErrorAction SilentlyContinue |
        Sort-Object Date -Descending |
        Select-Object -First 1

    if ($null -ne $mostRecent) {
        $daysSinceUpdate = ((Get-Date) - $mostRecent.Date).Days
        $lastUpdateTitle = $mostRecent.Title
        $recencyCheck    = if ($daysSinceUpdate -le $ml1ThresholdDays) { "PASS" } else { "FAIL" }
    } else {
        $recencyCheck = "WARN"
    }
}

# [5] EOL scan
$eolApplications = @(
    "Adobe Flash Player", "Internet Explorer", "Java 6", "Java 7",
    "Java(TM) 6", "Java(TM) 7", "Microsoft Silverlight",
    "Microsoft Visual C++ 2005", "Microsoft Visual C++ 2008",
    "Python 2", "7-Zip 1", "7-Zip 2"
)

$registryPaths = @(
    "HKLM:\Software\Microsoft\Windows\CurrentVersion\Uninstall\*",
    "HKLM:\Software\Wow6432Node\Microsoft\Windows\CurrentVersion\Uninstall\*"
)

$installedApps = $registryPaths | ForEach-Object {
    Get-ItemProperty -Path $_ -ErrorAction SilentlyContinue
} | Where-Object { $_.DisplayName }

$foundEol = @()
foreach ($eolEntry in $eolApplications) {
    $foundEol += $installedApps | Where-Object { $_.DisplayName -like "*$eolEntry*" }
}
$eolCheck = if ($foundEol.Count -eq 0) { "PASS" } else { "FAIL" }

# [6] Reboot pending
$rebootPending = (Test-Path "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Component Based Servicing\RebootPending") -or
                 (Test-Path "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\WindowsUpdate\Auto Update\RebootRequired")
$rebootCheck = if ($rebootPending) { "WARN" } else { "No" }

# Overall: any FAIL or WARN is non-compliant
$allChecks     = @($pswuCheck, $wingetCheck, $outstandingCheck, $recencyCheck, $eolCheck)
$overallStatus = if ($allChecks -contains "FAIL" -or $allChecks -contains "WARN") {
    "NON-COMPLIANT"
} else {
    "COMPLIANT"
}

$separator = "=" * 60

Write-Output $separator
Write-Output "  ESSENTIAL EIGHT ML1 -- PATCH APPLICATIONS"
Write-Output "  ASD Control: Mitigation Strategy 2"
Write-Output "  Reference  : cyber.gov.au/essential-eight"
Write-Output $separator
Write-Output "  Server     : $hostname"
Write-Output "  Report Date: $reportDate"
Write-Output "  Generated  : AttuneOps"
Write-Output $separator
Write-Output "  CHECKS"
Write-Output "  [1] PSWindowsUpdate Module"
Write-Output "     Result : $pswuCheck"
Write-Output "  [2] winget"
Write-Output "     Result : $wingetCheck"
Write-Output "  [3] Outstanding Application Security Updates"
Write-Output "     Pending: $($outstandingUpdates.Count)"
Write-Output "     Result : $outstandingCheck"
Write-Output "  [4] Application Update Recency (ISM-1691)"
Write-Output "     Most recent update     : $lastUpdateTitle"
Write-Output "     Days since last update : $(if ($daysSinceUpdate -ge 0) { $daysSinceUpdate } else { 'Unknown' })"
Write-Output "     ML1 threshold          : $ml1ThresholdDays days"
Write-Output "     Result : $recencyCheck"
Write-Output "  [5] EOL Applications (ISM-1704)"
Write-Output "     EOL apps found : $($foundEol.Count)"
Write-Output "     Result : $eolCheck"
if ($foundEol.Count -gt 0) {
    $foundEol | ForEach-Object {
        Write-Output "     EOL: $($_.DisplayName) -- version $($_.DisplayVersion)"
    }
}
Write-Output "  [6] Reboot Pending"
Write-Output "     Status : $rebootCheck"
Write-Output $separator
Write-Output "  OVERALL STATUS : $overallStatus"
Write-Output $separator

if ($overallStatus -eq "NON-COMPLIANT") {
    exit 1
}
