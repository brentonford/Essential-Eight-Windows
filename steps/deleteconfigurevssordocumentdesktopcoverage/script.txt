$os       = Get-CimInstance -ClassName Win32_OperatingSystem
$isServer = $os.Caption -match "Windows Server"

if ($isServer) {
    # wbadmin uses VSS automatically -- no manual VSS policy configuration is needed.
    # This step confirms the VSS service is healthy and the schedule from Step 3 is present.
    $vssSvc = Get-Service -Name VSS -ErrorAction SilentlyContinue
    if ($null -eq $vssSvc) {
        Write-Error "VSS service not found."
        exit 1
    }

    if ($vssSvc.Status -ne "Running") {
        Start-Service -Name VSS -ErrorAction SilentlyContinue
    }
    Set-Service -Name VSS -StartupType Automatic
    Write-Output "VSS service   : $((Get-Service VSS).Status), startup Automatic (ISM-1810)"

    # Confirm a wbadmin schedule exists by checking the WSB registry key.
    # wbadmin get schedule is not a valid subcommand -- WSB stores schedule config
    # under HKLM:\SYSTEM\CurrentControlSet\Services\wbengine\SystemStateBackup.
    $wsbScheduleKey = "HKLM:\SOFTWARE\Microsoft\Windows NT\CurrentVersion\SPP\Clients\{09F7EDC5-294E-4180-AF6A-FB0E6A0E9513}"
    $wsbTaskPath    = "\Microsoft\Windows\WindowsBackup\AutomaticBackup"
    $task = Get-ScheduledTask -TaskPath "\Microsoft\Windows\Backup\" -ErrorAction SilentlyContinue |
            Where-Object { $_.TaskName -match "WindowsBackup" } |
            Select-Object -First 1

    if ($null -eq $task) {
        Write-Output "WARN: No Windows Backup scheduled task found. Run Step 3 to configure it."
        exit 1
    }

    Write-Output "Backup schedule task  : $($task.TaskName) ($($task.State))"
    Write-Output "VSS and backup schedule verified."
    exit 0
}
