$mfaProviders = @(
    @{ Name = "Duo Security Authentication Proxy"; TestPath = "$env:ProgramFiles\Duo Security Authentication Proxy"; RegPath = "HKLM:\SOFTWARE\Duo Security" },
    @{ Name = "Azure MFA NPS Extension";           TestPath = "$env:ProgramFiles\Microsoft\AzureMfa";              RegPath = "HKLM:\SOFTWARE\Microsoft\AzureMfa" },
    @{ Name = "Windows Hello for Business (GPO)";  TestPath = $null;                                               RegPath = "HKLM:\SOFTWARE\Policies\Microsoft\PassportForWork" },
    @{ Name = "NPS Role (RADIUS server)";          TestPath = $null;                                               RegPath = "HKLM:\SYSTEM\CurrentControlSet\Services\IAS" }
)

$detectedProviders = @()

foreach ($provider in $mfaProviders) {
    $found = $false
    if ($null -ne $provider.TestPath -and (Test-Path $provider.TestPath)) { $found = $true }
    if (-not $found -and $null -ne $provider.RegPath -and (Test-Path $provider.RegPath)) { $found = $true }

    $status = if ($found) { "DETECTED" } else { "Not found" }
    Write-Output "  $($provider.Name) : $status"
    if ($found) { $detectedProviders += $provider.Name }
}

Write-Output ""
if ($detectedProviders.Count -gt 0) {
    Write-Output "MFA provider detected: $($detectedProviders -join ', ')"
} else {
    Write-Output "No MFA provider detected. RDP sessions are single-factor only."
    exit 1
}
