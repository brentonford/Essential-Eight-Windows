$os       = Get-CimInstance -ClassName Win32_OperatingSystem
$isServer = $os.Caption -match "Windows Server"

if ($isServer) {
    Import-Module WindowsServerBackup

    $policy = Get-WBPolicy
    Set-WBVssBackupOption -Policy $policy -VssBackupOptions VssCopyBackup
    Set-WBPolicy -Policy $policy
    Write-Output "VSS backup option set: VssCopyBackup (application-consistent -- ISM-1810)"

    $vssSvc = Get-Service -Name VSS
    if ($vssSvc.Status -ne "Running") { Start-Service -Name VSS }
    Set-Service -Name VSS -StartupType Automatic
    Write-Output "VSS service: $((Get-Service VSS).Status), startup Automatic"
    exit 0
}

# Desktop: document the backup coverage statement for ISM-1511
$oneDriveRoot  = (Get-ItemProperty -Path "HKCU:\SOFTWARE\Microsoft\OneDrive" -Name "UserFolder" -ErrorAction SilentlyContinue).UserFolder
$tenantName    = (Get-ItemProperty -Path "HKCU:\SOFTWARE\Microsoft\OneDrive\Accounts\Business1" -Name "DisplayName" -ErrorAction SilentlyContinue).DisplayName
$userEmail     = (Get-ItemProperty -Path "HKCU:\SOFTWARE\Microsoft\OneDrive\Accounts\Business1" -Name "UserEmail" -ErrorAction SilentlyContinue).UserEmail

Write-Output "======================================"
