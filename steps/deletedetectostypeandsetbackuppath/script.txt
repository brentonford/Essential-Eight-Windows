$os       = Get-CimInstance -ClassName Win32_OperatingSystem
$isServer = $os.Caption -match "Windows Server"

Write-Output "OS Caption : $($os.Caption)"
Write-Output "OS Type    : $(if ($isServer) { 'Windows Server -- using Windows Server Backup' } else { 'Windows Desktop -- using sync coverage verification' })"
Write-Output ""

if ($isServer) {
    Write-Output "Server backup path: Windows Server Backup to UNC destination."
    Write-Output "Destination: {backupDestinationPath}"
    Write-Output "Retention  : {backupRetentionDays} days"
} else {
    Write-Output "Desktop backup path: OneDrive/network sync coverage verification."
    Write-Output "Windows Server Backup steps will be skipped for this OS type."
}
