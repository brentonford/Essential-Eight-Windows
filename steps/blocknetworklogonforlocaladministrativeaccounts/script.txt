# ISM-1175 (partial) + Pass-the-Hash mitigation.
# Conditionally adds S-1-5-114 (all local accounts that are Administrators) to:
# - SeDenyNetworkLogonRight (deny access to this computer from the network)
# - SeDenyRemoteInteractiveLogonRight (deny log on through Remote Desktop Services)
#
# applyPthMitigation defaults to false because S-1-5-114 in SeDenyNetworkLogonRight
# breaks WinRM-based agentless management (AttuneOps) when local admin accounts
# are the WinRM credential. Set to true only if domain accounts are used for WinRM,
# or if local admin WinRM access on this host is explicitly not required.

$applyPthMitigation = $false

if ($applyPthMitigation) {
    $seceditExport = "$env:TEMP\secpol_pth.cfg"
    $seceditDb     = "$env:TEMP\secpol_pth.sdb"
    $seceditImport = "$env:TEMP\secpol_pth_new.cfg"

    secedit /export /cfg $seceditExport /quiet

    $content = Get-Content $seceditExport -Raw

    if ($content -match 'SeDenyNetworkLogonRight') {
        $content = $content -replace 'SeDenyNetworkLogonRight\s*=.*', 'SeDenyNetworkLogonRight = *S-1-5-32-546,*S-1-5-114'
    } else {
        $content += "`r`nSeDenyNetworkLogonRight = *S-1-5-32-546,*S-1-5-114"
    }

    if ($content -match 'SeDenyRemoteInteractiveLogonRight') {
        $content = $content -replace 'SeDenyRemoteInteractiveLogonRight\s*=.*', 'SeDenyRemoteInteractiveLogonRight = *S-1-5-32-546,*S-1-5-114'
    } else {
        $content += "`r`nSeDenyRemoteInteractiveLogonRight = *S-1-5-32-546,*S-1-5-114"
    }

    $content | Set-Content $seceditImport
    secedit /configure /db $seceditDb /cfg $seceditImport /areas USER_RIGHTS /quiet
    Remove-Item $seceditExport, $seceditImport, $seceditDb -Force -ErrorAction SilentlyContinue

    Write-Output "ISM-1175 full PtH mitigation applied."
    Write-Output "  SeDenyNetworkLogonRight           : Guest + S-1-5-114 (local admin accounts)"
    Write-Output "  SeDenyRemoteInteractiveLogonRight : Guest + S-1-5-114 (local admin accounts)"
} else {
    Write-Output "ISM-1175 partial implementation -- S-1-5-114 local PtH mitigation NOT applied."
    Write-Output "Compliance posture:"
    Write-Output "  Remote administration of this host originates from a Secure Admin"
    Write-Output "  Workstation (ISM-1380). Internet and email restrictions for privileged"
    Write-Output "  accounts (ISM-1175 full) are enforced at the SAW and network layer."
    Write-Output "  The local SeDenyNetworkLogonRight S-1-5-114 mitigation is intentionally"
    Write-Output "  not applied because it is incompatible with the WinRM-based agentless"
    Write-Output "  management model used to administer this server."
    Write-Output "Compensating controls:"
    Write-Output "  - ISM-1380: SAW used for all administrative connections to this host."
    Write-Output "  - Network segmentation between SAW and managed hosts."
    Write-Output "  - Audit logging of privileged sessions (see Step 6 of this Blueprint)."
    Write-Output "  - Named admin accounts only (built-in Administrator disabled in Step 2)."
    Write-Output "To opt in to the local PtH mitigation (and accept the loss of WinRM-based"
    Write-Output "remote management with local admin accounts on this host), set the"
    Write-Output "applyPthMitigation parameter to 'true' and re-run the Blueprint."
}
