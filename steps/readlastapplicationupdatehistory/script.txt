# ISM-1691 (ML1): patches for office productivity suites, web browsers, email clients,
# PDF software, and security products must be applied within 2 weeks of release.
# ISM-1876 (ML1): critical patches or those with working exploits within 48 hours.
# Threshold: 14 days.

$ml1ThresholdDays = 14

$module = Get-Module -ListAvailable -Name PSWindowsUpdate
if ($null -eq $module) {
    Write-Error "CHECK Update History : FAIL -- PSWindowsUpdate not available"
    exit 1
}

Import-Module PSWindowsUpdate

$history = Get-WUHistory -ErrorAction SilentlyContinue |
    Sort-Object Date -Descending |
    Select-Object -First 10

if ($null -eq $history -or $history.Count -eq 0) {
    Write-Error "CHECK Update History : WARN -- no update history found"
    exit 1
}

$mostRecent          = $history | Select-Object -First 1
$daysSinceLastUpdate = ((Get-Date) - $mostRecent.Date).Days

Write-Output "Most recent update     : $($mostRecent.Title)"
Write-Output "Date applied           : $($mostRecent.Date.ToString('yyyy-MM-dd HH:mm:ss'))"
Write-Output "Days since last update : $daysSinceLastUpdate"
Write-Output "Recent update history (last 10):"
$history | ForEach-Object {
    Write-Output "  [$($_.Date.ToString('yyyy-MM-dd'))] $($_.ResultCode) -- $($_.Title)"
}

if ($daysSinceLastUpdate -gt $ml1ThresholdDays) {
    Write-Error "CHECK Update History : FAIL -- last update $daysSinceLastUpdate days ago (ML1 threshold: $ml1ThresholdDays days)"
    exit 1
}

Write-Output "CHECK Update History : PASS -- last update within ${ml1ThresholdDays}-day ML1 threshold"
