$hostname   = $env:COMPUTERNAME
$reportDate = Get-Date -Format "yyyy-MM-dd HH:mm:ss"
$os         = Get-CimInstance -ClassName Win32_OperatingSystem
$isServer   = $os.Caption -match "Windows Server"

$archivePath        = "{policyExportPath}"
$policyName         = "E8-ML1-AppControl"
$policyXmlPath      = Join-Path $archivePath "$policyName.xml"
$activePoliciesPath = "C:\Windows\System32\CodeIntegrity\CiPolicies\Active"

# --- Deployment check ---
$xmlExists     = Test-Path $policyXmlPath
$policyId      = ""
$binaryDeployed = $false

if ($xmlExists) {
    [xml]$xml      = Get-Content -Path $policyXmlPath -Raw
    $policyId      = $xml.SiPolicy.PolicyID
    $policyIdClean = $policyId -replace '[{}]', ''
    $binaryDeployed = Test-Path (Join-Path $activePoliciesPath "{$policyIdClean}.cip")
}

$deployedCheck = if ($xmlExists -and $binaryDeployed) { "PASS" } elseif ($xmlExists -and -not $binaryDeployed) { "WARN -- reboot required" } else { "FAIL" }

# --- Enforcement mode check ---
$umciEnabled  = $false
$auditEnabled = $false

if ($xmlExists) {
    $ruleOptions = $xml.SelectNodes("//*[local-name()='Rule']/*[local-name()='Option']")
    foreach ($opt in $ruleOptions) {
        switch ($opt.InnerText) {
            "Enabled:UMCI"       { $umciEnabled  = $true }
            "Enabled:Audit Mode" { $auditEnabled = $true }
        }
    }
}

$modeCheck = if (-not $umciEnabled) {
    "FAIL -- UMCI not set"
} elseif ($auditEnabled) {
    "WARN -- Audit mode (not yet enforcing)"
} else {
    "PASS -- Enforce mode"
}

# --- Path deny rule check ---
$requiredPaths = @(
    "%OSDrive%\Users\*\AppData\Local\Temp\*",
    "%OSDrive%\Users\*\AppData\Local\*",
    "%OSDrive%\Users\*\AppData\Roaming\*",
    "%OSDrive%\Users\*\Downloads\*"
)

$coveredPaths  = @()
$denyRuleCount = 0

if ($xmlExists) {
    $denyRules = $xml.SelectNodes("//*[local-name()='FilePathRule'][@Action='Deny']")
    $denyRuleCount = $denyRules.Count
    foreach ($rule in $denyRules) {
        $fpNode = $rule.SelectSingleNode("*[local-name()='FileAttributes']/*[local-name()='FilePath']")
        if ($null -ne $fpNode) {
            $coveredPaths += $fpNode.GetAttribute("FilePath")
        }
    }
}

$missingPaths = $requiredPaths | Where-Object { $_ -notin $coveredPaths }
$pathCheck    = if ($missingPaths.Count -eq 0) { "PASS" } else { "FAIL -- $($missingPaths.Count) path(s) missing" }

# --- Event log check ---
$events = Get-WinEvent -FilterHashtable @{
    LogName   = "Microsoft-Windows-CodeIntegrity/Operational"
    Id        = @(3076, 3077, 3089)
    StartTime = (Get-Date).AddDays(-7)
} -ErrorAction SilentlyContinue

$auditCount   = @($events | Where-Object { $_.Id -eq 3076 }).Count
$enforceCount = @($events | Where-Object { $_.Id -eq 3077 }).Count
$policyCount  = @($events | Where-Object { $_.Id -eq 3089 }).Count

# --- ISM scope ---
$ismScope   = if ($isServer) { "ML2 Prep (ISM-1490 internet-facing / ISM-1656 non-internet-facing servers)" } else { "ML1 (ISM-0843 workstations + ISM-1870 user-writable path control)" }
$ismNote    = if ($isServer) { "NOTE: Application control on servers is ML2/ML3. This Blueprint implements ahead of that requirement." } else { "ISM-1870: Blocks execution from user profiles and temp folders." }

# --- Overall status ---
$failedChecks = @($deployedCheck, $modeCheck, $pathCheck) | Where-Object { $_ -like "FAIL*" }
$warnChecks   = @($deployedCheck, $modeCheck, $pathCheck) | Where-Object { $_ -like "WARN*" }

$overallStatus = if ($failedChecks.Count -gt 0) {
    "NON-COMPLIANT"
} elseif ($warnChecks.Count -gt 0) {
    "COMPLIANT WITH WARNINGS"
} else {
    "COMPLIANT"
}

$separator = "=" * 60

Write-Output $separator
Write-Output "  ESSENTIAL EIGHT -- APPLICATION CONTROL (WDAC)"
Write-Output "  ASD Control: Mitigation Strategy 1"
Write-Output "  Scope: $ismScope"
Write-Output "  Reference: cyber.gov.au/essential-eight"
Write-Output $separator
Write-Output "  Endpoint    : $hostname"
Write-Output "  OS          : $($os.Caption)"
Write-Output "  Report Date : $reportDate"
Write-Output "  Generated   : AttuneOps"
Write-Output $separator
Write-Output ""
Write-Output "  $ismNote"
Write-Output ""
Write-Output "  CHECKS"
Write-Output ""
Write-Output "  [1] Policy Deployed"
Write-Output "    Archive XML     : $xmlExists"
Write-Output "    Policy ID       : $policyId"
Write-Output "    Binary in Active: $binaryDeployed"
Write-Output "    Result          : $deployedCheck"
Write-Output ""
Write-Output "  [2] Enforcement Mode"
Write-Output "    UMCI enabled    : $umciEnabled"
Write-Output "    Audit mode      : $auditEnabled"
Write-Output "    Result          : $modeCheck"
Write-Output ""
Write-Output "  [3] Path Deny Rules (ISM-1870)"
Write-Output "    Deny rules found: $denyRuleCount"
Write-Output "    Required paths  : $($requiredPaths.Count)"
Write-Output "    Missing paths   : $($missingPaths.Count)"
Write-Output "    Result          : $pathCheck"
if ($missingPaths.Count -gt 0) {
    $missingPaths | ForEach-Object { Write-Output "      Missing: $_" }
}
Write-Output ""
Write-Output "  [4] WDAC Event Activity (last 7 days)"
Write-Output "    3076 Audit blocks   : $auditCount"
Write-Output "    3077 Enforce blocks : $enforceCount"
Write-Output "    3089 Policy events  : $policyCount"
if ($policyCount -eq 0 -and -not $binaryDeployed) {
    Write-Output "    NOTE: No policy events detected -- confirm server has been rebooted."
}
Write-Output ""
Write-Output $separator
Write-Output "  OVERALL STATUS : $overallStatus"
Write-Output $separator

if ($overallStatus -ne "COMPLIANT") {
    Write-Output ""
    if ($deployedCheck -like "WARN*") {
        Write-Output "  ACTION: Reboot the server to activate the deployed WDAC policy."
    }
    if ($modeCheck -like "WARN*") {
        Write-Output "  ACTION: Review Event ID 3076 audit blocks, then re-run Configure"
        Write-Output "  Blueprint with WDAC Enforcement Mode = 'Enforce' when ready."
    }
    if ($pathCheck -like "FAIL*") {
        Write-Output "  ACTION: Re-run Configure Blueprint to restore missing deny rules."
    }
    exit 1
}
