$hostname   = $env:COMPUTERNAME
$reportDate = Get-Date -Format "yyyy-MM-dd HH:mm:ss"
$os         = Get-CimInstance -ClassName Win32_OperatingSystem
$isServer   = $os.Caption -match "Windows Server"

$archivePath        = "{policyExportPath}"
$policyName         = "E8-ML1-AppControl"
$policyXmlPath      = Join-Path $archivePath "$policyName.xml"
$activePoliciesPath = "C:\Windows\System32\CodeIntegrity\CiPolicies\Active"

# [1] Policy deployment check
$xmlExists      = Test-Path $policyXmlPath
$policyId       = ""
$binaryDeployed = $false

if ($xmlExists) {
    [xml]$xml      = Get-Content -Path $policyXmlPath -Raw
    $policyId      = $xml.SiPolicy.PolicyID
    $policyIdClean = $policyId -replace '[{}]', ''
    $binaryPath    = Join-Path $activePoliciesPath "{$policyIdClean}.cip"
    $binaryDeployed = Test-Path $binaryPath
}

# [2] Enforcement mode check
$auditEnabled = $false
$umciEnabled  = $false
if ($xmlExists) {
    $ruleOptions = $xml.SelectNodes("//*[local-name()='Rule']/*[local-name()='Option']")
    foreach ($opt in $ruleOptions) {
        if ($opt.InnerText -eq "Enabled:UMCI")      { $umciEnabled  = $true }
        if ($opt.InnerText -eq "Enabled:Audit Mode") { $auditEnabled = $true }
    }
}

# [3] Path deny rules check
# <Deny FilePath="..." /> -- FilePath is a direct attribute. %OSDRIVE% all caps.
$requiredPaths = @(
    "%OSDRIVE%\Users\*\AppData\Local\Temp\*",
    "%OSDRIVE%\Users\*\AppData\Local\*",
    "%OSDRIVE%\Users\*\AppData\Roaming\*",
    "%OSDRIVE%\Users\*\Downloads\*"
)
$deployedPaths = [System.Collections.Generic.List[string]]::new()

if ($xmlExists) {
    $ns        = "urn:schemas-microsoft-com:sipolicy"
    $nsManager = New-Object System.Xml.XmlNamespaceManager($xml.NameTable)
    $nsManager.AddNamespace("si", $ns)

    $denyRules = $xml.SelectNodes("//si:FileRules/si:Deny", $nsManager)
    foreach ($rule in $denyRules) {
        $filePath = $rule.GetAttribute("FilePath")
        if (-not [string]::IsNullOrEmpty($filePath)) {
            $deployedPaths.Add($filePath)
        }
    }
}
$missingPaths = $requiredPaths | Where-Object { $_ -notin $deployedPaths }
$pathRulesOk  = $missingPaths.Count -eq 0

# [4] Event log counts
$events       = Get-WinEvent -FilterHashtable @{
    LogName   = "Microsoft-Windows-CodeIntegrity/Operational"
    Id        = @(3076, 3077)
    StartTime = (Get-Date).AddDays(-7)
} -ErrorAction SilentlyContinue
$auditCount   = @($events | Where-Object { $_.Id -eq 3076 }).Count
$enforceCount = @($events | Where-Object { $_.Id -eq 3077 }).Count

# Determine check results
$deployedCheck = if ($binaryDeployed) { "PASS" } else { "FAIL" }
$modeCheck     = if (-not $umciEnabled) { "FAIL" } elseif ($auditEnabled) { "WARN (Audit)" } else { "PASS (Enforce)" }
$pathCheck     = if ($pathRulesOk) { "PASS" } else { "FAIL" }

$overallStatus = if ($deployedCheck -eq "FAIL" -or $pathCheck -eq "FAIL" -or -not $umciEnabled) {
    "NON-COMPLIANT"
} elseif ($auditEnabled) {
    "COMPLIANT (Audit Mode -- switch to Enforce when ready)"
} else {
    "COMPLIANT"
}

$ismScope  = if ($isServer) {
    "ML2 Prep (ISM-1490 internet-facing / ISM-1656 non-internet-facing servers)"
} else {
    "ML1 (ISM-0843 + ISM-1870 workstations)"
}
$ismDetail = if ($isServer) {
    "NOTE: Application control on servers is ML2/ML3. This Blueprint implements ahead of that requirement."
} else {
    "ISM-0843: Application control on workstations. ISM-1870: Blocks user profile and temp folder execution."
}

$separator = "=" * 60
Write-Output $separator
Write-Output "  ESSENTIAL EIGHT -- APPLICATION CONTROL (WDAC)"
Write-Output "  ASD Control: Mitigation Strategy 1"
Write-Output "  Scope: $ismScope"
Write-Output "  Reference  : cyber.gov.au/essential-eight"
Write-Output $separator
Write-Output "  Endpoint   : $hostname"
Write-Output "  OS         : $($os.Caption)"
Write-Output "  Report Date: $reportDate"
Write-Output "  Generated  : AttuneOps"
Write-Output $separator
Write-Output "  $ismDetail"
Write-Output "  CHECKS"
Write-Output "  [1] Policy Deployed"
Write-Output "    XML archived     : $xmlExists"
Write-Output "    Policy ID        : $policyId"
Write-Output "    Binary in Active : $binaryDeployed"
Write-Output "    Result           : $deployedCheck"
Write-Output "  [2] Enforcement Mode"
Write-Output "    UMCI enabled  : $umciEnabled"
Write-Output "    Audit mode    : $auditEnabled"
Write-Output "    Result        : $modeCheck"
Write-Output "  [3] Path Deny Rules (ISM-1870)"
Write-Output "    Rules deployed : $($deployedPaths.Count)"
Write-Output "    Required paths : $($requiredPaths.Count)"
Write-Output "    Missing paths  : $($missingPaths.Count)"
Write-Output "    Result         : $pathCheck"
if ($missingPaths.Count -gt 0) {
    $missingPaths | ForEach-Object { Write-Output "    Missing: $_" }
}
Write-Output "  [4] WDAC Event Activity (last 7 days)"
Write-Output "    Audit blocks   (3076): $auditCount"
Write-Output "    Enforce blocks (3077): $enforceCount"
Write-Output $separator
Write-Output "  OVERALL STATUS : $overallStatus"
Write-Output $separator

if ($overallStatus -ne "COMPLIANT" -and $overallStatus -notmatch "Audit Mode") {
    exit 1
}
