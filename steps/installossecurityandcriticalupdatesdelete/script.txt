# ISM-1694 / ISM-1695 / ISM-1877
# Installs Windows OS-level security and critical updates via PSWindowsUpdate.
#
# Scope is deliberately restricted to OS-level update categories:
#   - Security Updates  (OS kernel, components, drivers)
#   - Critical Updates  (high-priority OS fixes)
#
# Explicitly excluded:
#   - "Application" category (covered by E2 Patch Applications Blueprint)
#   - "Upgrades" / Feature Updates (major OS version changes — require a
#     deliberate upgrade process, not an automated patch run)
#
# IgnoreReboot: servers must reboot in a controlled maintenance window.
# The pending reboot is reported in the Verify Blueprint.

Import-Module PSWindowsUpdate

Write-Output "Checking for available OS security and critical updates..."
Write-Output "Timestamp: $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')"
Write-Output ""

$excludeTitlePattern = "Feature update|Upgrade to"
$categories          = @("Security Updates", "Critical Updates")

foreach ($category in $categories) {
    $available = Get-WindowsUpdate `
        -Category $category `
        -NotTitle $excludeTitlePattern `
        -ErrorAction SilentlyContinue

    if ($null -eq $available -or $available.Count -eq 0) {
        Write-Output "Category '$category': no updates available."
        continue
    }

    Write-Output "Category '$category': $($available.Count) update(s) found:"
    $available | ForEach-Object {
        Write-Output "  [$($_.MsrcSeverity)] $($_.Title)"
    }
    Write-Output ""
    Write-Output "Installing '$category' updates via scheduled task (elevation required)..."

    $scriptBlock = "Import-Module PSWindowsUpdate; Install-WindowsUpdate -Category '$category' -NotTitle '$excludeTitlePattern' -AcceptAll -IgnoreReboot -Confirm:`$false"
    $taskName    = "AttuneOps-WU-$($category -replace ' ','')"
    $logPath     = "C:\E8\wupdate-$($category -replace ' ','').log"

    $action    = New-ScheduledTaskAction -Execute "powershell.exe" -Argument "-NonInteractive -NoProfile -Command `"$scriptBlock | Tee-Object -FilePath '$logPath'`""
    $principal = New-ScheduledTaskPrincipal -UserId "SYSTEM" -RunLevel Highest
    $settings  = New-ScheduledTaskSettingsSet -ExecutionTimeLimit (New-TimeSpan -Hours 2)

    Unregister-ScheduledTask -TaskName $taskName -Confirm:$false -ErrorAction SilentlyContinue
    Register-ScheduledTask -TaskName $taskName -Action $action -Principal $principal -Settings $settings | Out-Null
    Start-ScheduledTask -TaskName $taskName

    Write-Output "Scheduled task '$taskName' started. Waiting for completion..."

    $timeout = 120
    $elapsed = 0
    do {
        Start-Sleep -Seconds 10
        $elapsed += 10
        $state = (Get-ScheduledTask -TaskName $taskName).State
    } while ($state -eq "Running" -and $elapsed -lt $timeout * 60)

    if ($state -ne "Ready") {
        Write-Output "WARNING: Task '$taskName' ended in unexpected state: $state"
    }

    if (Test-Path $logPath) {
        Write-Output "--- Install log for '$category' ---"
        Get-Content $logPath | ForEach-Object { Write-Output $_ }
        Write-Output "--- End log ---"
    }

    Unregister-ScheduledTask -TaskName $taskName -Confirm:$false -ErrorAction SilentlyContinue
    Write-Output "Category '$category': installation complete."
    Write-Output ""
}

Write-Output "OS update installation run completed."
Write-Output "Timestamp: $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')"
Write-Output "Check Step 6 in the Verify Blueprint for pending reboot status."
