$rdpTcpPath = "HKLM:\SYSTEM\CurrentControlSet\Control\Terminal Server\WinStations\RDP-Tcp"
$rdpGpoPath = "HKLM:\SOFTWARE\Policies\Microsoft\Windows NT\Terminal Services"

$tcpNla   = (Get-ItemProperty -Path $rdpTcpPath -Name "UserAuthentication" -ErrorAction SilentlyContinue).UserAuthentication
$gpoNla   = (Get-ItemProperty -Path $rdpGpoPath -Name "UserAuthentication" -ErrorAction SilentlyContinue).UserAuthentication
$tcpSsl   = (Get-ItemProperty -Path $rdpTcpPath -Name "SecurityLayer"      -ErrorAction SilentlyContinue).SecurityLayer

$nlaActive = $tcpNla -eq 1 -or $gpoNla -eq 1

Write-Output "RDP-Tcp UserAuthentication (NLA) : $tcpNla"
Write-Output "GPO UserAuthentication (NLA)     : $gpoNla"
Write-Output "RDP-Tcp SecurityLayer (TLS)      : $tcpSsl"
Write-Output ""
Write-Output "CHECK NLA : $(if ($nlaActive) { 'PASS -- NLA is enabled' } else { 'FAIL -- NLA is not enabled' })"
Write-Output "NOTE: NLA is a prerequisite for MFA integration, not MFA itself."

if (-not $nlaActive) {
    exit 1
}
