# Reads Deny rules from the policy XML and verifies all ISM-1870 required
# paths are present.
#
# WDAC deny path rules are flat self-closing <Deny> elements with a FilePath
# attribute -- the same structure as <Allow> rules. This matches exactly how
# Step 4 of the Configure Blueprint writes them.
#
# Path variable is %OSDRIVE% (all caps) -- matches the Configure Blueprint.

$archivePath   = "{policyExportPath}"
$policyName    = "E8-ML1-AppControl"
$policyXmlPath = Join-Path $archivePath "$policyName.xml"

if (-not (Test-Path $policyXmlPath)) {
    Write-Error "CHECK Path Deny Rules : FAIL -- policy XML not found"
    exit 1
}

$requiredPaths = @(
    "%OSDRIVE%\Users\*\AppData\Local\Temp\*",
    "%OSDRIVE%\Users\*\AppData\Local\*",
    "%OSDRIVE%\Users\*\AppData\Roaming\*",
    "%OSDRIVE%\Users\*\Downloads\*"
)

[xml]$xml  = Get-Content -Path $policyXmlPath -Raw
$ns        = "urn:schemas-microsoft-com:sipolicy"
$nsManager = New-Object System.Xml.XmlNamespaceManager($xml.NameTable)
$nsManager.AddNamespace("si", $ns)

# <Deny FilePath="..." /> -- FilePath is a direct attribute on the Deny element
$denyRules     = $xml.SelectNodes("//si:FileRules/si:Deny", $nsManager)
$deployedPaths = [System.Collections.Generic.List[string]]::new()

foreach ($rule in $denyRules) {
    $filePath = $rule.GetAttribute("FilePath")
    if (-not [string]::IsNullOrEmpty($filePath)) {
        $deployedPaths.Add($filePath)
        Write-Output "Deny rule found: $filePath"
    }
}

Write-Output "Deny rules found: $($deployedPaths.Count)"

$missingPaths = $requiredPaths | Where-Object { $_ -notin $deployedPaths }

if ($missingPaths.Count -eq 0) {
    Write-Output "CHECK Path Deny Rules : PASS -- all ISM-1870 paths covered"
    exit 0
}

Write-Output "CHECK Path Deny Rules : FAIL -- $($missingPaths.Count) required path(s) missing:"
$missingPaths | ForEach-Object { Write-Output "  Missing: $_" }
exit 1
