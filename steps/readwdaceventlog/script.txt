# Event IDs in Microsoft-Windows-CodeIntegrity/Operational:
#   3076 = Audit block (would have been blocked in enforce mode)
#   3077 = Enforce block (was actively blocked)
#   3089 = Policy refresh/deployment event

$logName  = "Microsoft-Windows-CodeIntegrity/Operational"
$lookback = (Get-Date).AddDays(-7)

$events = Get-WinEvent -FilterHashtable @{
    LogName   = $logName
    Id        = @(3076, 3077, 3089)
    StartTime = $lookback
} -ErrorAction SilentlyContinue

if ($null -eq $events -or $events.Count -eq 0) {
    Write-Output "No WDAC events (3076/3077/3089) found in the last 7 days."
    Write-Output "Possible reasons:"
    Write-Output "- No execution attempts occurred in the blocked paths during this period"
    Write-Output "- Policy was deployed very recently and the system is still clean"
    exit 0
}

$auditBlocks   = @($events | Where-Object { $_.Id -eq 3076 })
$enforceBlocks = @($events | Where-Object { $_.Id -eq 3077 })
$policyEvents  = @($events | Where-Object { $_.Id -eq 3089 })

Write-Output "WDAC events (last 7 days):"
Write-Output "  Audit blocks   (3076): $($auditBlocks.Count)"
Write-Output "  Enforce blocks (3077): $($enforceBlocks.Count)"
Write-Output "  Policy events  (3089): $($policyEvents.Count)"

if ($auditBlocks.Count -gt 0) {
    Write-Output "Recent audit block events (last 5):"
    $auditBlocks | Sort-Object TimeCreated -Descending | Select-Object -First 5 | ForEach-Object {
        $ts      = $_.TimeCreated.ToString('yyyy-MM-dd HH:mm:ss')
        $message = $_.Message -replace '\s+', ' '
        Write-Output "  [$ts] $message"
    }
}

if ($enforceBlocks.Count -gt 0) {
    Write-Output "Recent enforce block events (last 5):"
    $enforceBlocks | Sort-Object TimeCreated -Descending | Select-Object -First 5 | ForEach-Object {
        Write-Output "  [$($_.TimeCreated.ToString('yyyy-MM-dd HH:mm:ss'))] Block event recorded"
    }
}
