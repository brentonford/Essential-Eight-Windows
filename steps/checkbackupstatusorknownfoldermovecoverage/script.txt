$os       = Get-CimInstance -ClassName Win32_OperatingSystem
$isServer = $os.Caption -match "Windows Server"

if (-not $isServer) {
    # Desktop path -- check OneDrive Known Folder Move
    $kfmPath = "HKCU:\Software\Microsoft\OneDrive\Accounts"
    $kfmOk   = $false

    if (Test-Path $kfmPath) {
        $accounts = Get-ChildItem $kfmPath -ErrorAction SilentlyContinue
        foreach ($account in $accounts) {
            $kfmDesktop   = (Get-ItemProperty $account.PSPath -Name "KFMFolderDesktop"   -ErrorAction SilentlyContinue).KFMFolderDesktop
            $kfmDocuments = (Get-ItemProperty $account.PSPath -Name "KFMFolderDocuments" -ErrorAction SilentlyContinue).KFMFolderDocuments
            if ($kfmDesktop -and $kfmDocuments) {
                $kfmOk = $true
                break
            }
        }
    }

    Write-Output "Known Folder Move (Desktop)   : $(if ($kfmOk) { 'Configured' } else { 'Not configured' })"
    Write-Output "Known Folder Move (Documents) : $(if ($kfmOk) { 'Configured' } else { 'Not configured' })"
    Write-Output "CHECK Known Folder Move : $(if ($kfmOk) { 'PASS' } else { 'FAIL -- KFM not configured' })"
    exit $(if ($kfmOk) { 0 } else { 1 })
}

# Server path -- requires Windows Server Backup module
Import-Module WindowsServerBackup -ErrorAction SilentlyContinue

if (-not (Get-Command Get-WBSummary -ErrorAction SilentlyContinue)) {
    Write-Output "CHECK Backup Status : FAIL -- WindowsServerBackup module unavailable (feature not installed?)"
    exit 1
}

$summary = Get-WBSummary -ErrorAction SilentlyContinue

if ($null -eq $summary) {
    Write-Output "CHECK Backup Status : FAIL -- no backup summary available (no backups run yet?)"
    exit 1
}

$lastSuccess = $summary.LastSuccessfulBackupTime
$ageHours    = if ($null -ne $lastSuccess) {
    [math]::Round((New-TimeSpan -Start $lastSuccess -End (Get-Date)).TotalHours, 1)
} else {
    $null
}

Write-Output "Last successful backup  : $(if ($null -ne $lastSuccess) { $lastSuccess.ToString('yyyy-MM-dd HH:mm:ss') } else { 'Never' })"
Write-Output "Backup age (hours)      : $(if ($null -ne $ageHours) { $ageHours } else { 'N/A' })"
Write-Output "Last backup result      : $($summary.LastBackupResultHR)"

$ageOk = $null -ne $ageHours -and $ageHours -le 24

Write-Output "CHECK Backup Status : $(if ($ageOk) { 'PASS' } else { 'FAIL -- last backup older than 24 hours or never run' })"
exit $(if ($ageOk) { 0 } else { 1 })
