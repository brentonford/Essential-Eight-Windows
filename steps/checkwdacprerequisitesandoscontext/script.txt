$os          = Get-CimInstance -ClassName Win32_OperatingSystem
$buildNumber = [int]$os.BuildNumber
$isServer    = $os.Caption -match "Windows Server"

Write-Output "OS Caption    : $($os.Caption)"
Write-Output "Build Number  : $buildNumber"
Write-Output "OS Type       : $(if ($isServer) { 'Server' } else { 'Desktop/Workstation' })"

# Multi-policy format requires Windows 10 1903 (build 18362) or Server 2019 (build 17763)
$multiPolicyMinBuild = 17763

if ($buildNumber -lt $multiPolicyMinBuild) {
    Write-Error "This Blueprint requires Windows 10 1903+, Windows Server 2019+, or later."
    Write-Error "Build $buildNumber is below the minimum required build $multiPolicyMinBuild."
    exit 1
}

$configCI = Get-Module -ListAvailable -Name ConfigCI
if ($null -eq $configCI) {
    Write-Error "ConfigCI module not found. Ensure Windows Defender Application Control tools are available."
    exit 1
}

Write-Output "ConfigCI module     : Available"
Write-Output "Multi-policy format : Supported (build $buildNumber >= $multiPolicyMinBuild)"

if ($isServer) {
    Write-Output "ISM SCOPE: This server will implement WDAC ahead of ML2 requirements."
    Write-Output "ISM-1490 (ML2): Application control on internet-facing servers."
    Write-Output "ISM-1656 (ML3): Application control on non-internet-facing servers."
    Write-Output "ML1 application control (ISM-0843) applies to workstations only."
} else {
    Write-Output "ISM SCOPE: ML1 compliance."
    Write-Output "ISM-0843 (ML1): Application control on workstations."
    Write-Output "ISM-1870 (ML1): Applied to user profiles and temporary folders."
}
