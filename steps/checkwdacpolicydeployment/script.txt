$archivePath        = "{policyExportPath}"
$policyName         = "E8-ML1-AppControl"
$policyXmlPath      = Join-Path $archivePath "$policyName.xml"
$activePoliciesPath = "C:\Windows\System32\CodeIntegrity\CiPolicies\Active"

Write-Output "Policy XML archive path  : $policyXmlPath"
Write-Output "Active policies directory: $activePoliciesPath"

$xmlExists = Test-Path $policyXmlPath
Write-Output "Policy XML present in archive : $xmlExists"

if (-not $xmlExists) {
    Write-Error "CHECK Policy Deployed : FAIL -- policy XML not found in archive"
    exit 1
}

[xml]$xml       = Get-Content -Path $policyXmlPath -Raw
$policyId       = $xml.SiPolicy.PolicyID
$policyIdClean  = $policyId -replace '[{}]', ''
$binaryFileName = "{$policyIdClean}.cip"
$activeBinaryPath = Join-Path $activePoliciesPath $binaryFileName

Write-Output "Policy ID (from XML) : $policyId"
Write-Output "Expected binary      : $activeBinaryPath"

if (-not (Test-Path $activeBinaryPath)) {
    Write-Error "CHECK Policy Deployed : FAIL -- binary not found in active policies directory"
    exit 1
}

$binary = Get-Item $activeBinaryPath
Write-Output "Binary present        : True"
Write-Output "Binary size           : $($binary.Length) bytes"
Write-Output "Binary last modified  : $($binary.LastWriteTime.ToString('yyyy-MM-dd HH:mm:ss'))"
Write-Output "CHECK Policy Deployed : PASS"
